Usage example

Download the manifest and place it beside this Python script to read its sample records. Code is not executed by this page.

import json

with open("axonhub-demo-manifest.json", encoding="utf-8") as file:
    manifest = json.load(file)
dataset = next(resource for resource in manifest["resources"] if resource["type"] == "dataset")
for row in dataset["sample"]:
    print(row["language"], row["sentiment"], row["text"])
